VIAGVVT0 (per ogni giorno guardare anche il giorno precedente e seguente alla DTRIF)
Alla fine delle query controllare comunque "a occhio" tutti i record con le DTRIF modificate
--------
	*** DTINC e DTINV
	UPDATE    dbo.VIAGVVT0
	SET              VVTDTINC = CONVERT(DATETIME, '2008-11-06 00:00:00', 102), VVTDTINV = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)
	WHERE     (VVTDTRIF = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)) AND (VVTDTINC = CONVERT(DATETIME, '2008-06-11 00:00:00', 102))
	
	UPDATE    dbo.VIAGVVT0
	SET              VVTDTINC = CONVERT(DATETIME, '2008-11-05 00:00:00', 102), VVTDTINV = CONVERT(DATETIME, '2008-11-05 00:00:00', 102)
	WHERE     (VVTDTRIF = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)) AND (VVTDTINC = CONVERT(DATETIME, '2008-05-11 00:00:00', 102))
	
	UPDATE    dbo.VIAGVVT0
	SET              VVTDTINC = CONVERT(DATETIME, '2008-11-07 00:00:00', 102), VVTDTINV = CONVERT(DATETIME, '2008-11-07 00:00:00', 102)
	WHERE     (VVTDTRIF = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)) AND (VVTDTINC = CONVERT(DATETIME, '2008-07-11 00:00:00', 102))



VIAGVVC0 (anche qui per ogni giorno vanno guardate le date precendti e seguenti alla DTCON)
Alla fine delle query controllare comunque "a occhio" tutti i record con le DTCON modificate
--------
	*** DTARR
	UPDATE    dbo.VIAGVVC0
	SET              VVCDTARR = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)
	WHERE     (VVCDTCON = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)) AND (VVCDTARR = CONVERT(DATETIME, '2008-06-11 00:00:00', 102))
	
	UPDATE    dbo.VIAGVVC0
	SET              VVCDTARR = CONVERT(DATETIME, '2008-11-07 00:00:00', 102)
	WHERE     (VVCDTCON = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)) AND (VVCDTARR = CONVERT(DATETIME, '2008-07-11 00:00:00', 102))
	
	UPDATE    dbo.VIAGVVC0
	SET              VVCDTARR = CONVERT(DATETIME, '2008-11-05 00:00:00', 102)
	WHERE     (VVCDTCON = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)) AND (VVCDTARR = CONVERT(DATETIME, '2008-05-11 00:00:00', 102))

	*** DTPRE
	UPDATE    dbo.VIAGVVC0
	SET              VVCDTARR = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)
	WHERE     (VVCDTCON = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)) AND (VVCDTARR = CONVERT(DATETIME, '2008-06-11 00:00:00', 102))
	
	UPDATE    dbo.VIAGVVC0
	SET              VVCDTARR = CONVERT(DATETIME, '2008-11-05 00:00:00', 102)
	WHERE     (VVCDTCON = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)) AND (VVCDTARR = CONVERT(DATETIME, '2008-05-11 00:00:00', 102))
	
	UPDATE    dbo.VIAGVVC0
	SET              VVCDTARR = CONVERT(DATETIME, '2008-11-07 00:00:00', 102)
	WHERE     (VVCDTCON = CONVERT(DATETIME, '2008-11-06 00:00:00', 102)) AND (VVCDTARR = CONVERT(DATETIME, '2008-07-11 00:00:00', 102))
	
	
	
Date viaggi spostate indietro
---------
	SELECT     VVTIDVVT, VVTDTINC, VVTDTINV, VVTDTFIV, VVTDTRIF, VVTFLAGG
	FROM         dbo.VIAGVVT0
	WHERE     (VVTDTRIF > CONVERT(DATETIME, '2008-10-01 00:00:00', 102)) AND (ABS(DATEDIFF([day], VVTDTRIF, VVTDTINV)) > 2)
